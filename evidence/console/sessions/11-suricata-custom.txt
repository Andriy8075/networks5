# Lab5 custom rules for lr5_v02.pcap
# Thresholds relative to baseline of legitimate clients:
# ~60-80 new TCP connections/min (~1-2/s) and ~1 HTTP request per connection.
# Attack rates: ~1000 SYN in ~1.5s (scan), ~165 HTTP req/s (flood), 10 ICMP in <1s (recon).

# 1) SYN port scan (nmap-like): SYN with window 1024, burst from one source
# window:1024 matches the scanner fingerprint and avoids HTTP-flood SYNs (window 64240)
alert tcp any any -> $HOME_NET any (msg:"LAB5 SYN Port Scan"; flags:S,12; window:1024; threshold:type both, track by_src, count 50, seconds 5; classtype:attempted-recon; sid:1000001; rev:2;)

# 2) HTTP flood: burst of GET /?<digits> toward web server
# threshold 80/2s >> baseline (~1-2 req/s from all legit clients combined)
alert http any any -> $HOME_NET 80 (msg:"LAB5 HTTP Flood GET numeric query"; flow:to_server,established; http.method; content:"GET"; http.uri; content:"/?"; pcre:"/\/\?\d+/"; threshold:type both, track by_dst, count 80, seconds 2; classtype:denial-of-service; sid:1000002; rev:1;)

# 3) ICMP host discovery: rapid echo requests from one source
# baseline: almost no ICMP from clients; recon sends 10 echoes in <1s
alert icmp any any -> $HOME_NET any (msg:"LAB5 ICMP Host Sweep"; itype:8; threshold:type both, track by_src, count 5, seconds 2; classtype:attempted-recon; sid:1000003; rev:1;)
