## Контрольна сума і capinfos
sha256sum lr5_v02.pcap
capinfos lr5_v02.pcap
tshark -r lr5_v02.pcap -T fields -e frame.time_utc | head -n 1
tshark -r lr5_v02.pcap -T fields -e frame.time_utc | tail -n 1

## Статистика tshark
tshark -r lr5_v02.pcap -q -z io,phs
tshark -r lr5_v02.pcap -q -z conv,ip
tshark -r lr5_v02.pcap -q -z endpoints,ip
tshark -r lr5_v02.pcap -q -z io,stat,10
tshark -r lr5_v02.pcap -q -z "io,stat,1,tcp.flags.syn==1 && tcp.flags.ack==0"
